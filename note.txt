### create RAS key
mkdir -p .local/keys
New-Item -ItemType Directory -Force .\local-keys
openssl genpkey -algorithm RSA `
  -pkeyopt rsa_keygen_bits:2048 `
  -out .\local-keys\jwt-private.pem

openssl rsa `
  -pubout `
  -in .\local-keys\jwt-private.pem `
  -out .\local-keys\jwt-public.pem
